# 洛谷P1955 [NOI2015] 程序自动分析
没什么好说的，就是一个离散化加一个并查集，然后发现我的离散化非常不熟练  
有一个很诡异的地方是离散化明明是让数据更集中但是叫**离散**化  
代码：  

```cpp
#include<bits/stdc++.h>
using namespace std;
int fa[200010],siz[200010]={0},t,n,tmp[200010];
map<int,int>f;
struct proc
{
    int x,y,t;
}a[100010];
int find(int x)
{
    if(fa[x]==x)return x;
    return fa[x]=find(fa[x]);
}
void ad(int x,int y)
{
    int fx=find(x),fy=find(y);
    if(fx==fy)return;
    if(siz[fx]<siz[fy])swap(fx,fy);
    if(siz[fx]==siz[fy])siz[fy]++;
    fa[fx]=fy;
}
int main()
{
    cin>>t;
    for(int k=1;k<=t;k++)
    {
        cin>>n;
        for(int i=1;i<=n;i++)
        {
            cin>>a[i].x>>a[i].y>>a[i].t;
            tmp[2*i-1]=a[i].x;tmp[2*i]=a[i].y;

        }
        sort(tmp+1,tmp+2*n+1);
        int m=unique(tmp+1,tmp+2*n+1)-tmp-1;
        for(int i=1;i<=m;i++)
        {
            f[tmp[i]]=i;
        }
        for(int i=1;i<=2*n;i++)
        {
            fa[i]=i;
            siz[i]=1;
        }
        for(int i=1;i<=n;i++)
        {
            if(a[i].t==1)
            {
                ad(f[a[i].x],f[a[i].y]);
            }
            
        }
        bool flag=1;
        for(int i=1;i<=n;i++)
        {
            if(a[i].t==0)
            {
                if(find(f[a[i].x])==find(f[a[i].y]))
                {
                    cout<<"NO"<<endl;
                    flag=0;
                    break;
                }
            }
            
        }
        if(flag)cout<<"YES"<<endl;
    }
}
```
