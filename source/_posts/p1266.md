# **洛谷P1266 [BalticOI 2002] 速度限制**
## **1.审题**
这道题肯定不能一上来就最短路，因为对于没有限速的路，他的**长度**是要受走过的上一条路影响的，不能简单递推。想到的解法是把以一种速度到达一个城市的状态当成一个节点，这样的话每两个节点之间的距离是清晰的
## **2.代码**
``` cpp
#include<bits/stdc++.h>
#define double float//不然会mle
using namespace std;
struct node
{
    double dis;int u;
    bool operator<(const node& rhs)const
    {
        return dis>rhs.dis;
    }
};
struct edge
{
    int v;double w;
};
vector<edge>e[100010];
int n,m,d;
double dis[100010];
int la[100010];
bool flag[100010]={0};
priority_queue<node>q;
void dij()
{
    for(int i=1;i<=100009;i++)dis[i]=1e9;
    dis[151*70]=0;
    q.push({0,151*70});
    while(!q.empty())
    {
        node tmp=q.top();
        q.pop();
        if(flag[tmp.u])continue;
        flag[tmp.u]=1;
        for(int i=0;i<e[tmp.u].size();i++)
        {
            if(dis[e[tmp.u][i].v]>dis[tmp.u]+e[tmp.u][i].w)
            {
                dis[e[tmp.u][i].v]=dis[tmp.u]+e[tmp.u][i].w;
                la[e[tmp.u][i].v]=tmp.u;
                q.push({dis[e[tmp.u][i].v],e[tmp.u][i].v});
            }
        }
    }
}
void search(int x)
{
    if(x%151==0)
    {
        cout<<0<<' ';
        return;
    }
    search(la[x]);
    cout<<x%151<<' ';
    return;
}
int main()
{
    cin>>n>>m>>d;
    for(int i=1;i<=m;i++)
    {
        int tmp1,tmp2;double tmp3,tmp4;
        cin>>tmp1>>tmp2>>tmp3>>tmp4;
        if(tmp3==0)
        {
            for(int j=0;j<=500;j++)
            {
                e[tmp1+151*j].push_back({int(tmp2+151*j),tmp4/double(j)});
            }
        }
        else
        {
            for(int j=0;j<=500;j++)
            {
                e[tmp1+151*j].push_back({int(tmp2+151*tmp3),tmp4/tmp3});
            }
        }
    }
    dij();
    double min=1e9;int idx=0;
    for(int i=1;i<=500;i++)
    {
        if(dis[d+151*i]<min)
        {
            min=dis[d+151*i];
            idx=d+151*i;
        }
    }
    search(idx);
}
