# 洛谷P2921 [USACO08DEC] Trick or Treat on the Farm G 复杂度纠错
## **1.AC代码1**
这是我的第一版代码，核心思路是找出每一个连通块的圈（容易证明是存在唯一的）把圈上的点处理好后，再遍历一遍点，让不在圈上的走到圈上从而计算需要走的步数。以这个思路去写代码的话理论上是能做到o(n)的。
``` cpp
#include<bits/stdc++.h>
using namespace std;
int t[100010],ans[100010]={0},n;
bool flag1[100010]={0};
int main()
{
    cin>>n;
    for(int i=1;i<=n;i++)
    {
        cin>>t[i];
    }
    for(int i=1;i<=n;i++)
    {
        if(flag1[i])continue;
        int tmp=i;
        int flag2[100010]={0};//问题1
        while(flag1[tmp]==0)
        {
            flag1[tmp]=1;
            flag2[tmp]=1;
            tmp=t[tmp];
        }
        if(flag2[tmp]==1)
        {
            int cnt=0;
            int ttmp=t[tmp];
            while(ttmp!=tmp)
            {
                ttmp=t[ttmp];
                cnt++;
            }
            for(int j=1;j<=cnt+1;j++)
            {
                ans[tmp]=cnt+1;
                tmp=t[tmp];
            }
        }
    }
    for(int i=1;i<=n;i++)//问题2
    {
        if(ans[i]!=0)continue;
        int cnt=0;
        int tmp=i;
        while(ans[tmp]==0)
        {
            cnt++;
            tmp=t[tmp];
        }
        ans[i]=cnt+ans[tmp];
    }
    for(int i=1;i<=n;i++)cout<<ans[i]<<endl;
}
```
## **2.问题** 
实际上这段代码的最坏复杂度是$o(n^2)$:
- 在问题1处的初始化相当于每次循环要做100010次操作
- 在问题二处的循环最坏情况下可以是$o(n^2)$的  

但是由于数据没卡和编译器优化等等原因，这个代码是能过的
## **3.正确（？）代码**
``` cpp
#include<bits/stdc++.h>
using namespace std;
int t[100010],ans[100010]={0},n,cnt=0,flag2[100010];
bool flag1[100010]={0};
void dfs(int x)
{
    if(ans[x])return;
    dfs(t[x]);
    ans[x]=ans[t[x]]+1;
    return;
}
int main()
{
    cin>>n;
    for(int i=1;i<=n;i++)
    {
        cin>>t[i];
    }
    for(int i=1;i<=n;i++)
    {
        if(flag1[i])continue;
        int tmp=i;
        cnt++;
        while(flag1[tmp]==0)
        {
            flag1[tmp]=1;
            flag2[tmp]=cnt;//这样实际上不用每次初始化
            tmp=t[tmp];
        }
        if(flag2[tmp]==cnt)
        {
            int cnt=0;
            int ttmp=t[tmp];
            while(ttmp!=tmp)
            {
                ttmp=t[ttmp];
                cnt++;
            }
            for(int j=1;j<=cnt+1;j++)
            {
                ans[tmp]=cnt+1;
                tmp=t[tmp];
            }
        }
    }
    for(int i=1;i<=n;i++)
    {
        dfs(i);//这里的dfs是记忆化的所以复杂度是o(n)
    }
    for(int i=1;i<=n;i++)cout<<ans[i]<<endl;
}
```
